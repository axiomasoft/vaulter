## Item spec
### P1.2 — Инварианты дерева

**Intent:** Дерево каждого storage остаётся корректным при любых вызовах PHP API и HTTP и при параллельных операциях:
родитель/назначение — активная папка того же storage, корень неприкосновенен, у storage ровно один узел без родителя,
циклы невозможны даже при гонке `A→B` ‖ `B→A`, имена нормализованы (N04, N05, AD13, I1–I4; V18, V19).

**Why:** OPUS 13 P1.2, 02 D13, 08 §1/§5, 09 §4. Сегодня ни одно Action не проверяет тип и состояние назначения
(`packages/drive/src/Actions/MoveNodeAction.php:37–66` проверяет только storage, цикл и имя — **до** транзакции и без
блокировок); `CreateDocumentNodeAction` пишет `parent_id = NULL` при `parent = null`
(`packages/documents/src/Actions/Documents/CreateDocumentNodeAction.php:45,62`) — «второй корень», невидимый в листинге
корня; `RestoreNodeAction`/`MoveNodeAction` с `into = null` тоже пишут `parent_id = NULL`.

**Scope Included:**
1. Исключения (контракт P1.8, `ProblemException`):
   - `packages/core/src/Exceptions/Nodes/InvalidDestinationException.php` — код `invalid_destination`, 409; фабрики
     `notAFolder(Node $destination)`, `trashed(Node $node)`, `rootIsImmutable(Node $root, string $operation)`;
   - `packages/core/src/Exceptions/Nodes/NodeNotTrashedException.php` — `node_not_trashed`, 409, `forNode(Node $node)`;
   - `packages/core/src/Exceptions/Nodes/InvalidNodeNameException.php` — `validation_failed`, 422,
     `renderExtras()` = `['errors' => ['name' => [$this->getMessage()]]]`, фабрика `forName(string $name, string $reason)`.
   Межстораджевые операции бросают существующий `CrossStorageMoveException` (`cross_drive_operation`, 409).
2. `packages/core/src/Support/NodeName.php` — `final class NodeName { public static function normalize(string $name): string }`:
   trim пробелов по краям; NFC через `\Normalizer::normalize($name, \Normalizer::FORM_C)`; отказ (`InvalidNodeNameException`)

final class VaulterDatabase

    public function __construct(private readonly DatabaseManager $db) {}

    public function connection(): Connection
    {
        return $this->db->connection(Config::connection());
    }

    public function driver(): string
    {
        return $this->connection()->getDriverName();
    }

    /**
     * @template TReturn
     *
     * @param  Closure(Connection): TReturn $callback
     * @param  int<1, max>                  $attempts
     * @return TReturn
     */
    public function transaction(Closure $callback, int $attempts = 3): mixed
    {
        return $this->connection()->transaction($callback, $attempts);
    }

    public function lockTree(int|string $storageId): void
    {
        $this->requireTransaction('lockTree');

        if ($this->driver() === 'pgsql') {
            $this->advisoryLock('vaulter:tree:'.$storageId);

            return;
        }

        $this->connection()->table(Config::table('storages'))
            ->where('id', $storageId)->lockForUpdate()->first(['id']);
    }

    public function advisoryLock(string $key): void
    {
        $this->requireTransaction('advisoryLock');

        if ($this->driver() === 'pgsql') {
            $this->connection()->statement('SELECT pg_advisory_xact_lock(hashtext(?))', [$key]);
        }
    }

    public function afterCommit(Closure $callback): void
    {
        $this->connection()->afterCommit($callback);
    }

    private function requireTransaction(string $method): void
    {
        if ($this->connection()->transactionLevel() === 0) {
            throw new LogicException('[Vaulter] '.$method.'() must run inside VaulterDatabase::transaction().');
        }
    }
}

+<?php
+
+declare(strict_types=1);
+
+namespace Vaulter\Support;
+
+use Normalizer;
+use Vaulter\Exceptions\Nodes\InvalidNodeNameException;
+
+/**
+ * The canonical form of a node name, applied by every Action that writes one. A name
+ * is a single tree segment, so separators, control characters and the `.`/`..` path
+ * aliases are rejected rather than escaped. NFC keeps visually identical names (a
+ * precomposed `é` and `e` + combining accent) from coexisting as siblings; without
+ * ext-intl the name is kept as given.
+ */
+final class NodeName
+{
+    public static function normalize(string $name): string
+    {
+        if (!mb_check_encoding($name, 'UTF-8')) {
+            throw InvalidNodeNameException::forName($name, 'it is not valid UTF-8');
+        }
+
+        $name = trim($name, " \t\n\r\x0B");
+
+        if (class_exists(Normalizer::class)) {
+            $normalized = Normalizer::normalize($name, Normalizer::FORM_C);
+            $name = is_string($normalized) ? $normalized : $name;
+        }
+
+        if ($name === '') {
+            throw InvalidNodeNameException::forName($name, 'it is empty');
+        }
+
+        if ($name === '.' || $name === '..') {
+            throw InvalidNodeNameException::forName($name, 'it is a path alias');
+        }
+
+        if (preg_match('~[/\\\\]~', $name) === 1) {
+            throw InvalidNodeNameException::forName($name, 'it contains a path separator');
+        }
+
+        if (preg_match('/[\x{00}-\x{1F}\x{7F}-\x{9F}]/u', $name) === 1) {
+            throw InvalidNodeNameException::forName($name, 'it contains a control character');
+        }
+
+        $maxLength = Config::driveNameMaxLength();
+
+        if (strlen($name) > $maxLength) {
+            throw InvalidNodeNameException::forName($name, "it is longer than {$maxLength} bytes");
+        }
+
+        return $name;
+    }
+}

+<?php
+
+declare(strict_types=1);
+
+use Vaulter\Drive\Actions\MoveNodeAction;
+use Vaulter\Drive\Data\MoveNodeData;
+use Vaulter\Drive\Exceptions\CycleException;
+use Vaulter\Models\Drive\Node;
+use Vaulter\Tests\Concurrency\ForkRace;
+use Vaulter\Tests\Concurrency\RaceOutcome;
+use Vaulter\Tests\Drive\DriveTestHelpers;
+
+/**
+ * Two concurrent moves that would each be legal alone but together close a loop.
+ * The cycle check reads the subtree, so it is only sound if no other structural
+ * change of the same drive commits between that read and the UPDATE — the drive
+ * tree lock provides that. Real concurrency needs PostgreSQL (ForkRace).
+ */
+beforeEach(function (): void {
+    ForkRace::skipUnlessSupported($this);
+});
+
+function raceMove(int|string $nodeId, int|string $intoId): Closure
+{
+    return static fn (): string => (string) app(MoveNodeAction::class)->execute(new MoveNodeData(
+        node: Node::query()->findOrFail($nodeId),
+        into: Node::query()->findOrFail($intoId),
+    ))->getKey();
+}
+
+/**
+ * Walk parent links up from the node; a tree reaches the root within the node count.
+ */
+function reachesRoot(int|string $nodeId, int|string $rootId): bool
+{
+    $current = $nodeId;
+
+    for ($step = 0; $step <= Node::query()->count(); $step++) {
+        if ((string) $current === (string) $rootId) {
+            return true;
+        }
+
+        $current = Node::query()->whereKey($current)->value('parent_id');
+
+        if ($current === null) {
+            return false;
+        }
+    }
+
+    return false;
+}
+
+/**
+ * @param list<RaceOutcome> $outcomes
+ */
+function assertOneMoveWins(array $outcomes): void
+{
+    $won = array_values(array_filter($outcomes, static fn (RaceOutcome $outcome): bool => $outcome->ok));
+    $lost = array_values(array_filter($outcomes, static fn (RaceOutcome $outcome): bool => !$outcome->ok));
+
+    expect($won)->toHaveCount(1)
+        ->and($lost)->toHaveCount(1)
+        ->and($lost[0]->failedWith(CycleException::class))->toBeTrue("the losing move failed with {$lost[0]->exception}: {$lost[0]->message}");
+}
+
+it('lets exactly one of move(A→B) and move(B→A) win and leaves no cycle', function (): void {
+    $drive = DriveTestHelpers::drive();
+    $rootId = $drive->root()->getKey();
+    $a = $drive->createFolder('A')->getKey();
+    $b = $drive->createFolder('B')->getKey();
+
+    assertOneMoveWins(ForkRace::run([raceMove($a, $b), raceMove($b, $a)]));
+
+    expect(reachesRoot($a, $rootId))->toBeTrue()
+        ->and(reachesRoot($b, $rootId))->toBeTrue();
+});
+
+// Row locks on the moved node and its destination alone serialise the two-node race
+// above; here the two moves lock disjoint rows, so only the tree lock keeps the
+// second cycle check from reading a subtree the first move is about to change.
+it('lets exactly one of move(A→B/B1) and move(B→A/A1) win and leaves no cycle', function (): void {
+    $drive = DriveTestHelpers::drive();
+    $rootId = $drive->root()->getKey();
+    $a = $drive->createFolder('A');
+    $b = $drive->createFolder('B');
+    $a1 = $drive->createFolder('A1', $a)->getKey();
+    $b1 = $drive->createFolder('B1', $b)->getKey();
+
+    assertOneMoveWins(ForkRace::run([raceMove($a->getKey(), $b1), raceMove($b->getKey(), $a1)]));
+
+    expect(reachesRoot($a1, $rootId))->toBeTrue()
+        ->and(reachesRoot($b1, $rootId))->toBeTrue();
+});
